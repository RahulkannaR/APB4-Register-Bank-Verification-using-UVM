     19  /// Address Decoder: Maps the input address combinatorially to an index.
     20  /// DYNamic Configuration (DYNC) version
     21  /// The address map `addr_map_i` is a packed array of rule_t structs.
     22  /// The ranges of any two rules may overlap. If so, the rule at the higher (more significant)
     23  /// position in `addr_map_i` prevails.
     24  ///
     25  /// There can be an arbitrary number of address rules. There can be multiple
     26  /// ranges defined for the same index. The start address has to be less than the end address.
     27  ///
     28  /// There is the possibility to add a default mapping:
     29  /// `en_default_idx_i`: Driving this port to `1'b1` maps all input addresses
     30  /// for which no rule in `addr_map_i` exists to the default index specified by
     31  /// `default_idx_i`. In this case, `dec_error_o` is always `1'b0`.
     32  ///
     33  /// The `Napot` parameter allows using naturally-aligned power of two (NAPOT) regions,
     34  /// using base addresses and masks instead of address ranges to specify rules.
     35  ///
     36  /// Assertions: The module checks every time there is a change in the address mapping
     37  /// if the resulting map is valid. It fatals if `start_addr` is higher than `end_addr` (non-NAPOT
     38  /// only) or if a mapping targets an index that is outside the number of allowed indices.
     39  /// It issues warnings if the address regions of any two mappings overlap (non-NAPOT only).
     40  module cc_addr_decode_dync #(
     41    /// Highest index which can happen in a rule.
     42    parameter int unsigned NoIndices = 32'd0,
     43    /// Total number of rules.
     44    parameter int unsigned NoRules   = 32'd1,
     45    /// Address type inside the rules and to decode.
     46    parameter type         addr_t    = logic,
     47    /// Whether this is a NAPOT (base and mask) or regular range decoder
     48    parameter bit          Napot     = 0,
     49    /// The output index type `idx_t` can be specified either with the width `IdxWidth`
     50    /// or directly with the type `idx_t`. By default, it will use the maximum index
     51    /// `NoIndices` to calculate the required width.
     52    parameter int unsigned IdxWidth  = cc_pkg::idx_width(NoIndices),
     53    parameter type         idx_t     = logic [IdxWidth-1:0],
     54    /// Rule packed struct type.
     55    /// The address decoder expects three fields in `rule_t`:
     56    ///
     57    /// typedef struct packed {
     58    ///   idx_t  idx;
     59    ///   addr_t start_addr;
     60    ///   addr_t end_addr;
     61    /// } rule_t;
     62    ///
     63    ///  - `idx`:        The index to be returned for a matching rule. Usually an integer but can
     64    ///                  be any type of data.
     65    ///  - `start_addr`: start address of the range the rule describes, value is included in range
     66    ///  - `end_addr`:   end address of the range the rule describes, value is NOT included in range
     67    ///                  if `end_addr == '0` end of address space is assumed
     68    ///
     69    /// If `Napot` is 1, The field names remain the same, but the rule describes a naturally-aligned
     70    /// power of two (NAPOT) region instead of an address range: `start_addr` becomes the base address
     71    /// and `end_addr` the mask. See the wrapping module `cc_addr_decode_napot` for details.
     72    // verilog_lint: waive typedef-structs-unions
     73    parameter type         rule_t    = struct packed {
     74      idx_t  idx;
     75      addr_t start_addr;
     76      addr_t end_addr;
     77    }
     78  ) (
     79    /// Address to decode.
     80    input  addr_t               addr_i,
     81    /// Address map: rule with the highest array position wins on collision
     82    input  rule_t [NoRules-1:0] addr_map_i,
     83    /// Decoded index.
     84    output idx_t                idx_o,
     85    /// Decode is valid.
     86    output logic                dec_valid_o,
     87    /// Decode is not valid, no matching rule found.
     88    output logic                dec_error_o,
     89    /// Enable default port mapping.
     90    ///
     91    /// When not used, tie to `0`.
     92    input  logic                en_default_idx_i,
     93    /// Default port index.
     94    ///
     95    /// When `en_default_idx_i` is `1`, this will be the index when no rule matches.
     96    ///
     97    /// When not used, tie to `0`.
     98    input  idx_t                default_idx_i,
     99    /// The module is dynamically configured, this deactivates its output and the integrated
     100    /// assertions.
     101    input  logic                config_ongoing_i
     102  );
     103  
     104    logic [NoRules-1:0] matched_rules; // purely for address map debugging
     105  
     106    always_comb begin
     107      // default assignments
     108      matched_rules = '0;
     109      dec_valid_o   = 1'b0;
     110      dec_error_o   = (en_default_idx_i) ? 1'b0 : 1'b1;
     111      idx_o         = (en_default_idx_i) ? default_idx_i : '0;
     112  
     113      // match the rules
     114      for (int unsigned i = 0; i < NoRules; i++) begin
     115        if (
     116          !Napot && (addr_i >= addr_map_i[i].start_addr) &&
     117          ((addr_i < addr_map_i[i].end_addr) || (addr_map_i[i].end_addr == '0)) ||
     118          Napot && (addr_map_i[i].start_addr & addr_map_i[i].end_addr) ==
     119                   (addr_i & addr_map_i[i].end_addr)
     120        ) begin
     121          matched_rules[i] = ~config_ongoing_i;
     122          dec_valid_o      = ~config_ongoing_i;
     123          dec_error_o      = 1'b0;
     124          idx_o            = config_ongoing_i ? default_idx_i : idx_t'(addr_map_i[i].idx);
     125        end
     126      end
     127    end
     128  
     129    // Assumptions and assertions
     130    `ifndef COMMON_CELLS_ASSERTS_OFF
     131    `ifndef SYNTHESIS
     132    initial begin : proc_check_parameters
     133      `ASSUME_I(addr_width_mismatch, $bits(addr_i) == $bits(addr_map_i[0].start_addr),
     134               $sformatf("Input address has %d bits and address map has %d bits.",
     135                         $bits(addr_i), $bits(addr_map_i[0].start_addr)))
     136      `ASSUME_I(norules_0, NoRules > 0, $sformatf("At least one rule needed"))
     137    end
     138  
     139    `ASSERT_FINAL(more_than_1_bit_set, $onehot0(matched_rules) || config_ongoing_i,
     140                  "More than one bit set in the one-hot signal, matched_rules")
     141  
     142    // These following assumptions check the validity of the address map.
     143    // The assumptions gets generated for each distinct pair of rules.
     144    // Each assumption is present two times, as they rely on one rules being
     145    // effectively ordered. Only one of the rules with the same function is
     146    // active at a time for a given pair.
     147    // check_start:        Enforces a smaller start than end address.
     148    // check_idx:          Enforces a valid index in the rule.
     149    // check_overlap:      Warns if there are overlapping address regions.
     150    always_comb begin : proc_check_addr_map
     151      if (!$isunknown(addr_map_i) && ~config_ongoing_i) begin
     152        for (int unsigned i = 0; i < NoRules; i++) begin
     153          `ASSUME_I(check_start, Napot || addr_map_i[i].start_addr < addr_map_i[i].end_addr ||
     154            addr_map_i[i].end_addr == '0,
     155            $sformatf("This rule has a higher start than end address!!!\n\
     156                Violating rule %d.\n\
     157                Rule> IDX: %h START: %h END: %h\n\
     158                #####################################################",
     159                i ,addr_map_i[i].idx, addr_map_i[i].start_addr, addr_map_i[i].end_addr))
     160          for (int unsigned j = i + 1; j < NoRules; j++) begin
     161            // overlap check
     162            `ASSUME_I(check_overlap, Napot ||
     163                                    !((addr_map_i[j].start_addr < addr_map_i[i].end_addr) &&
     164                                      (addr_map_i[j].end_addr > addr_map_i[i].start_addr)) ||
     165                                    !((addr_map_i[i].end_addr == '0) &&
     166                                      (addr_map_i[j].end_addr > addr_map_i[i].start_addr)) ||
     167                                    !((addr_map_i[j].start_addr < addr_map_i[i].end_addr) &&
     168                                      (addr_map_i[j].end_addr == '0)),
     169                $sformatf("Overlapping address region found!!!\n\
     170                Rule %d: IDX: %h START: %h END: %h\n\
     171                Rule %d: IDX: %h START: %h END: %h\n\
     172                #####################################################",
     173                i, addr_map_i[i].idx, addr_map_i[i].start_addr, addr_map_i[i].end_addr,
     174                j, addr_map_i[j].idx, addr_map_i[j].start_addr, addr_map_i[j].end_addr))
     175          end
     176        end
     177      end
     178    end
     179    `endif
     180    `endif
     181  
     182  endmodule
