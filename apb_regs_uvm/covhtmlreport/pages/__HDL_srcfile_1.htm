     11  // APB Read-Write Registers
     12  // Description: This module exposes a number of registers on an APB interface.
     13  //              It responds to not mapped accesses with a slave error.
     14  //              Some of the registers can be configured to be read only.
     15  // Parameters:
     16  // - `NoApbRegs`:    Number of registers.
     17  // - `ApbAddrWidth`: Address width of `req_i.paddr`, is used to generate internal address map.
     18  // - `AddrOffset`:   The address offset in bytes between the registers. Is asserted to be a least
     19  //                   `32'd4`, the reason is to prevent overlap in the address map.
     20  //                   Each register is mapped to a 4 byte wide address range. When this parameter
     21  //                   is bigger than `32'd4` there will be holes in the address map and the module
     22  //                   answers with `apb_pkg::RESP_SLVERR`. It is recommended that this value is a
     23  //                   power of 2, to prevent data alignment issues on upstream buses.
     24  // - `ApbDataWidth`: The data width of the APB4 bus, this value can be up to `32'd32` (bit).
     25  // - `RegDataWidth`: The data width of the registers, this value has to be less or equal to
     26  //                   `ApbDataWidth`. If it is less the register gets zero extended for reads and
     27  //                   higher bits on writes get ignored.
     28  // - `ReadOnly`:     This flag can specify a read only register at the given register index of the
     29  //                   array. When in the array the corresponding bit is set, the `reg_init_i` signal
     30  //                   at given index can be read out. Writes are ignored if the flag is set.
     31  // - `req_t`:        APB4 request struct. See macro definition in `include/typedef.svh`
     32  // - `resp_t`:       APB4 response struct. See macro definition in `include/typedef.svh`
     33  //
     34  // Ports:
     35  //
     36  // - `pclk_i:        Clock input signal (1-bit).
     37  // - `preset_ni:     Asynchronous active low reset signal (1-bit).
     38  // - `req_i:         APB4 request struct, bundles all APB4 signals from the master (req_t).
     39  // - `resp_o:        APB4 response struct, bundles all APB4 signals to the master (resp_t).
     40  // - `base_addr_i:   Base address of this module, from here the registers `0` are mapped, starting
     41  //                   with index `0`. All subsequent register with higher indices have their bases
     42  //                   mapped with reg_index * `AddrOffset` from this value (ApbAddrWidth-bit).
     43  // - `reg_init_i:    Initialization value for each register, when the register index is configured
     44  //                   as `ReadOnly[reg_index] == 1'b1` this value is passed through directly to
     45  //                   the APB4 bus (Array size `NoApbRegs` * RegDataWidth-bit).
     46  // - `reg_q_o:       The current value of the register. If the register at the array index is
     47  //                   read only, the `reg_init_i` value is passed through to the respective index
     48  //                   (Array size `NoApbRegs` * RegDataWidth-bit).
     49  //
     50  // This file also features the module `apb_regs_intf`. The difference is that instead of the
     51  // request and response structs it uses an `APB.Salve` interface. The parameters have the same
     52  // Function, however are defined in `ALL_CAPS`.
     53  
     54  
     55  `include "rtl/common_cells-master/include/common_cells/registers.svh"
     56  `include "rtl/common_cells-master/src/deprecated/cf_math_pkg.sv"
     57  `include "rtl/common_cells-master/src/deprecated/addr_decode.sv"
     58  
     59  module apb_regs #(
     60    parameter int unsigned        NoApbRegs    = 32'd0,
     61    parameter int unsigned        ApbAddrWidth = 32'd0,
     62    parameter int unsigned        AddrOffset   = 32'd4,
     63    parameter int unsigned        ApbDataWidth = 32'd0,
     64    parameter int unsigned        RegDataWidth = 32'd0,
     65    parameter bit [NoApbRegs-1:0] ReadOnly     = 32'h0,
     66    parameter type                req_t        = logic,
     67    parameter type                resp_t       = logic,
     68    // DEPENDENT PARAMETERS DO NOT OVERWRITE!
     69    parameter type apb_addr_t                  = logic[ApbAddrWidth-1:0],
     70    parameter type reg_data_t                  = logic[RegDataWidth-1:0]
     71  ) (
     72    // APB Interface
     73    input  logic                      pclk_i,
     74    input  logic                      preset_ni,
     75    input  req_t                      req_i,
     76    output resp_t                     resp_o,
     77    // Register Interface
     78    input  apb_addr_t                 base_addr_i, // base address of the read/write registers
     79    input  reg_data_t [NoApbRegs-1:0] reg_init_i,
     80    output reg_data_t [NoApbRegs-1:0] reg_q_o
     81  );
     82    localparam int unsigned IdxWidth  = (NoApbRegs > 32'd1) ? $clog2(NoApbRegs) : 32'd1;
     83    typedef logic [IdxWidth-1:0]     idx_t;
     84    typedef logic [ApbDataWidth-1:0] apb_data_t;
     85    typedef struct packed {
     86      int unsigned idx;
     87      apb_addr_t   start_addr;
     88      apb_addr_t   end_addr;
     89    } rule_t;
     90  
     91    logic has_reset_d, has_reset_q;
     92    `FFARN(has_reset_q, has_reset_q, 1'b0, pclk_i, preset_ni)
     93    assign has_reset_d = 1'b1;
     94  
     95    // signal declarations
     96    rule_t     [NoApbRegs-1:0] addr_map;
     97    idx_t                      reg_idx;
     98    logic                      decode_valid;
     99    // register signals
     100    reg_data_t [NoApbRegs-1:0] reg_d, reg_q;
     101    logic      [NoApbRegs-1:0] reg_update;
     102  
     103    // generate address map for the registers
     104    for (genvar i = 0; i < NoApbRegs; i++) begin: gen_reg_addr_map
     105      assign addr_map[i] = rule_t'{
     106        idx:        unsigned'(i),
     107        start_addr: base_addr_i + apb_addr_t'( i        * AddrOffset),
     108        end_addr:   base_addr_i + apb_addr_t'((i+32'd1) * AddrOffset)
     109      };
     110    end
     111  
     112    always_comb begin
     113      // default assignments
     114      reg_d      = has_reset_q ? reg_q : reg_init_i;
     115      reg_update = '0;
     116      resp_o     = '{
     117        pready:  req_i.psel & req_i.penable,
     118        prdata:  apb_data_t'(32'h0BAD_B10C),
     119        pslverr: apb_pkg::RESP_OKAY
     120      };
     121      if (req_i.psel) begin
     122        if (!decode_valid) begin
     123          // Error response on decode errors
     124          resp_o.pslverr = apb_pkg::RESP_SLVERR;
     125        end else begin
     126          if (req_i.pwrite) begin
     127            if (!ReadOnly[reg_idx]) begin
     128              for (int unsigned i = 0; i < RegDataWidth; i++) begin
     129                if (req_i.pstrb[i/8]) begin
     130                  reg_d[reg_idx][i] = req_i.pwdata[i];
     131                end
     132              end
     133              reg_update[reg_idx] = |req_i.pstrb;
     134            end else begin
     135              // this register is read only
     136              resp_o.pslverr = apb_pkg::RESP_SLVERR;
     137            end
     138          end else begin
     139            if (!ReadOnly[reg_idx]) begin
     140              resp_o.prdata = apb_data_t'(reg_q[reg_idx]);
     141            end else begin
     142              // for read only register the directly connect the init signal
     143              resp_o.prdata = apb_data_t'(reg_init_i[reg_idx]);
     144            end
     145          end
     146        end
     147      end
     148    end
     149  
     150    // output assignment and registers
     151    for (genvar i = 0; i < NoApbRegs; i++) begin : gen_rw_regs
     152      assign reg_q_o[i] = ReadOnly[i] ? reg_init_i[i] : reg_q[i];
     153      `FFLARN(reg_q[i], reg_d[i], reg_update[i], '0, pclk_i, preset_ni)
     154    end
     155  
     156    addr_decode #(
     157      .NoIndices ( NoApbRegs  ),
     158      .NoRules   ( NoApbRegs  ),
     159      .addr_t    ( apb_addr_t ),
     160      .rule_t    ( rule_t     )
     161    ) i_addr_decode (
     162      .addr_i      ( req_i.paddr  ),
     163      .addr_map_i  ( addr_map     ),
     164      .idx_o       ( reg_idx      ),
     165      .dec_valid_o ( decode_valid ),
     166      .dec_error_o ( /*not used*/ ),
     167      .en_default_idx_i ( '0      ),
     168      .default_idx_i    ( '0      )
     169    );
     170  
     171    // Validate parameters.
     172    // pragma translate_off
     173    `ifndef VERILATOR
     174      initial begin: p_assertions
     175        assert (NoApbRegs > 32'd0)
     176            else $fatal(1, "The number of registers must be at least 1!");
     177        assert (ApbAddrWidth > 32'd2)
     178            else $fatal(1, "ApbAddrWidth is not wide enough, has to be at least 3 bit wide!");
     179        assert (AddrOffset > 32'd3)
     180            else $fatal(1, "AddrOffset has to be at least 4 and is recommended to be a power of 2!");
     181        assert ($bits(req_i.paddr) == ApbAddrWidth)
     182            else $fatal(1, "AddrWidth does not match req_i.paddr!");
     183        assert (ApbDataWidth == $bits(resp_o.prdata))
     184            else $fatal(1, "ApbDataWidth has to be: ApbDataWidth == $bits(req_i.prdata)!");
     185        assert (ApbDataWidth > 32'd0 && ApbDataWidth <= 32'd32)
     186            else $fatal(1, "ApbDataWidth has to be: 32'd32 >= RegDataWidth > 0!");
     187        assert ($bits(resp_o.prdata) == $bits(req_i.pwdata))
     188            else $fatal(1, "req_i.pwdata has to match resp_o.prdata in width!");
     189        assert (RegDataWidth > 32'd0 && RegDataWidth <= 32'd32)
     190            else $fatal(1, "RegDataWidth has to be: 32'd32 >= RegDataWidth > 0!");
     191        assert (RegDataWidth <= $bits(resp_o.prdata))
     192            else $fatal(1, "RegDataWidth has to be: RegDataWidth <= $bits(req_i.prdata)!");
     193        assert (NoApbRegs == $bits(ReadOnly))
     194            else $fatal(1, "Each register need a `ReadOnly` flag!");
     195      end
     196    `endif
     197    // pragma translate_on
     198  endmodule

     200  `include "rtl/apb-master/include/apb/assign.svh"
     201  `include "rtl/apb-master/include/apb/typedef.svh"

     203  module apb_regs_intf #(
     204    parameter int unsigned          NO_APB_REGS    = 32'd0, // number of read only registers
     205    parameter int unsigned          APB_ADDR_WIDTH = 32'd0, // address width of `paddr`
     206    parameter int unsigned          ADDR_OFFSET    = 32'd4, // address offset in bytes
     207    parameter int unsigned          APB_DATA_WIDTH = 32'd0, // data width of the registers
     208    parameter int unsigned          REG_DATA_WIDTH = 32'd0,
     209    parameter bit [NO_APB_REGS-1:0] READ_ONLY      = 32'h0,
     210    // DEPENDENT PARAMETERS DO NOT OVERWRITE!
     211    parameter type                  apb_addr_t     = logic[APB_ADDR_WIDTH-1:0],
     212    parameter type                  reg_data_t     = logic[REG_DATA_WIDTH-1:0]
     213  ) (
     214    // APB Interface
     215    input  logic                        pclk_i,
     216    input  logic                        preset_ni,
     217    APB.Slave                           slv,
     218    // Register Interface
     219    input  apb_addr_t                   base_addr_i, // base address of the read only registers
     220    input  reg_data_t [NO_APB_REGS-1:0] reg_init_i,  // initalisation value for the registers
     221    output reg_data_t [NO_APB_REGS-1:0] reg_q_o
     222  );
     223    localparam int unsigned APB_STRB_WIDTH = cf_math_pkg::ceil_div(APB_DATA_WIDTH, 8);
     224    typedef logic [APB_DATA_WIDTH-1:0] apb_data_t;
     225    typedef logic [APB_STRB_WIDTH-1:0] apb_strb_t;
     226  
     227    `APB_TYPEDEF_REQ_T(apb_req_t, apb_addr_t, apb_data_t, apb_strb_t)
     228    `APB_TYPEDEF_RESP_T(apb_resp_t, apb_data_t)
     229  
     230    apb_req_t  apb_req;
     231    apb_resp_t apb_resp;
     232  
     233    `APB_ASSIGN_TO_REQ(apb_req, slv)
     234    `APB_ASSIGN_FROM_RESP(slv, apb_resp )
     235  
     236    apb_regs #(
     237      .NoApbRegs    ( NO_APB_REGS    ),
     238      .ApbAddrWidth ( APB_ADDR_WIDTH ),
     239      .AddrOffset   ( ADDR_OFFSET    ),
     240      .ApbDataWidth ( APB_DATA_WIDTH ),
     241      .RegDataWidth ( REG_DATA_WIDTH ),
     242      .ReadOnly     ( READ_ONLY      ),
     243      .req_t        ( apb_req_t      ),
     244      .resp_t       ( apb_resp_t     )
     245    ) i_apb_regs (
     246      .pclk_i,
     247      .preset_ni,
     248      .req_i       ( apb_req  ),
     249      .resp_o      ( apb_resp ),
     250      .base_addr_i,
     251      .reg_init_i,
     252      .reg_q_o
     253    );
     254  
     255    // Validate parameters.
     256    // pragma translate_off
     257    `ifndef VERILATOR
     258      initial begin: p_assertions
     259        assert (APB_ADDR_WIDTH == $bits(slv.paddr))
     260            else $fatal(1, "APB_ADDR_WIDTH does not match slv interface!");
     261        assert (APB_DATA_WIDTH == $bits(slv.pwdata))
     262            else $fatal(1, "APB_DATA_WIDTH does not match slv interface!");
     263      end
     264    `endif
     265    // pragma translate_on
     266  endmodule
