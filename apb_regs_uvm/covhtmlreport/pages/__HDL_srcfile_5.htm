1  //-----------------------------------------------------------------------------
     2  // tb_top.sv
     3  // Top-level testbench: clock/reset gen, apb_if instance, struct packing
     4  // glue between apb_if (flat signals) and apb_regs DUT (req_t/resp_t ports),
     5  // UVM config_db wiring, run_test() launch.
     6  //-----------------------------------------------------------------------------
     7  `timescale 1ns/1ps
     8  
     9  import uvm_pkg::*;
     10  import apb_uvm_pkg::*;
     11  `include "uvm_macros.svh"
     12  `include "apb_protocol_checker.sv"
     13  `include "reset_if.sv"
     14  
     15  module tb_top;
     16  
     17    //---------------------------------------------------------------------
     18    // Parameters  MUST match apb_reg_block / apb_scoreboard / apb_coverage
     19    //---------------------------------------------------------------------
     20    localparam int unsigned NoApbRegs    = 8;
     21    localparam int unsigned ApbAddrWidth = 12;
     22    localparam int unsigned AddrOffset   = 4;
     23    localparam int unsigned ApbDataWidth = 32;
     24    localparam int unsigned RegDataWidth = 32;
     25    localparam bit [NoApbRegs-1:0] ReadOnly = 8'b0000_0001;
     26    localparam logic [ApbAddrWidth-1:0] BaseAddr = 12'h000;
     27  
     28    //---------------------------------------------------------------------
     29    // Clock
     30    //---------------------------------------------------------------------
     31    logic pclk;
     32  
     33    initial pclk = 0;
     34    always #5 pclk = ~pclk; // 100MHz
     35  
     36    //---------------------------------------------------------------------
     37    // Reset  now test-controlled via reset_if / apb_reset_driver
     38    //---------------------------------------------------------------------
     39    reset_if rst_if();
     40    //wire preset_n = rst_if.preset_n;
     41  
     42    initial rst_if.preset_n = 1'b0; // starting state; base_test drives the real sequence
     43  
     44    //---------------------------------------------------------------------
     45    // APB interface instance
     46    //---------------------------------------------------------------------
     47    apb_if vif (.pclk(pclk), .preset_n(rst_if.preset_n));
     48  
     49    //---------------------------------------------------------------------
     50    // DUT req/resp struct wires + pack/unpack glue
     51    //---------------------------------------------------------------------
     52    apb_req_t  dut_req;
     53    apb_resp_t dut_resp;
     54  
     55    assign dut_req.paddr   = vif.paddr[ApbAddrWidth-1:0];
     56    assign dut_req.pprot   = vif.pprot;
     57    assign dut_req.psel    = vif.psel;
     58    assign dut_req.penable = vif.penable;
     59    assign dut_req.pwrite  = vif.pwrite;
     60    assign dut_req.pwdata  = vif.pwdata;
     61    assign dut_req.pstrb   = vif.pstrb;
     62  
     63    assign vif.pready  = dut_resp.pready;
     64    assign vif.prdata  = dut_resp.prdata;
     65    assign vif.pslverr = dut_resp.pslverr;
     66  
     67    //---------------------------------------------------------------------
     68    // Register-side ports (testbench drives init values, not RTL params)
     69    //---------------------------------------------------------------------
     70    logic [NoApbRegs-1:0][RegDataWidth-1:0] reg_init;
     71    logic [NoApbRegs-1:0][RegDataWidth-1:0] reg_q_out;
     72  
     73    genvar gi;
     74    generate
     75      for (gi = 0; gi < NoApbRegs; gi++) begin : g_reg_init
     76        assign reg_init[gi] = '0; // must match apb_reg_block reset values
     77      end
     78    endgenerate
     79  
     80    //---------------------------------------------------------------------
     81    // DUT instantiation
     82    //---------------------------------------------------------------------
     83    apb_regs #(
     84      .NoApbRegs    (NoApbRegs),
     85      .ApbAddrWidth (ApbAddrWidth),
     86      .AddrOffset   (AddrOffset),
     87      .ApbDataWidth (ApbDataWidth),
     88      .RegDataWidth (RegDataWidth),
     89      .ReadOnly     (ReadOnly),
     90      .req_t        (apb_req_t),
     91      .resp_t       (apb_resp_t)
     92    ) u_dut (
     93      .pclk_i      (pclk),
     94      .preset_ni   (rst_if.preset_n),
     95      .req_i       (dut_req),
     96      .resp_o      (dut_resp),
     97      .base_addr_i (BaseAddr),
     98      .reg_init_i  (reg_init),
     99      .reg_q_o     (reg_q_out)
     100    );
     101  
     102    apb_protocol_checker u_protocol_checker (.vif(vif));
     103  
     104    //---------------------------------------------------------------------
     105    // UVM config_db wiring + test launch
     106    //---------------------------------------------------------------------
     107    initial begin
     108      uvm_config_db#(virtual apb_if.DRIVER)::set(null, "*.agent.driver",  "vif", vif);
     109      uvm_config_db#(virtual apb_if.MONITOR)::set(null, "*.agent.monitor", "vif", vif);
     110      uvm_config_db#(virtual reset_if)::set(null, "*", "rst_vif", rst_if);
     111      run_test();
     112    end
     113  
     114    //---------------------------------------------------------------------
     115    // Optional: waveform dump for run.do
     116    //---------------------------------------------------------------------
     117    initial begin
     118      $dumpfile("apb_tb.vcd");
     119      $dumpvars(0, tb_top);
     120    end
     121  
     122  endmodule
