1  //-----------------------------------------------------------------------------
     2  // apb_protocol_checker.sv
     3  // SVA-based APB4 protocol legality checks. Bound to apb_if in tb_top.
     4  // This is the industry-standard way to check signal-timing legality 
     5  // continuous, cycle-by-cycle, independent of any specific test's traffic.
     6  //-----------------------------------------------------------------------------
     7  module apb_protocol_checker (apb_if.CHECKER vif);
     8  
     9    // 1. SETUP phase must be exactly 1 cycle: psel=1,penable=0 -> next cycle penable=1
     10    property p_setup_one_cycle;
     11      @(posedge vif.pclk) disable iff (!vif.preset_n)
     12      (vif.psel && !vif.penable) |=> vif.penable;
     13    endproperty
     14    a_setup_one_cycle: assert property (p_setup_one_cycle)
     15      else `uvm_error("PROTO_ERR", "SETUP phase did not transition to ACCESS next cycle")
     16  
     17    // 2. PADDR/PWDATA/PWRITE must stay stable from SETUP through ACCESS
     18    //    (until pready completes the transfer)
     19    property p_addr_stable;
     20      @(posedge vif.pclk) disable iff (!vif.preset_n)
     21      (vif.psel && !vif.penable) |-> (vif.paddr === $past(vif.paddr,0)) ##1
     22        (vif.penable && !vif.pready) |-> $stable(vif.paddr);
     23    endproperty
     24    // simplified stable-through-access check
     25    property p_paddr_pwdata_stable_in_access;
     26      @(posedge vif.pclk) disable iff (!vif.preset_n)
     27      (vif.psel && vif.penable && !vif.pready) |=> $stable(vif.paddr) && $stable(vif.pwdata);
     28    endproperty
     29    a_paddr_pwdata_stable: assert property (p_paddr_pwdata_stable_in_access)
     30      else `uvm_error("PROTO_ERR", "PADDR/PWDATA changed mid-ACCESS before PREADY")
     31  
     32    // 3. PENABLE must never be high in the same cycle PSEL first asserts (no skipping SETUP)
     33    property p_no_setup_skip;
     34      @(posedge vif.pclk) disable iff (!vif.preset_n)
     35      ($rose(vif.psel)) |-> !vif.penable;
     36    endproperty
     37    a_no_setup_skip: assert property (p_no_setup_skip)
     38      else `uvm_error("PROTO_ERR", "PENABLE high on same cycle PSEL first asserted  SETUP skipped")
     39  
     40    // 4. Consecutive transfer: SETUP is legal starting the cycle right after
     41    //    a completed ACCESS (psel may stay high)  this is a coverage-style
     42    //    check, not an error condition, so implemented as a cover property
     43    cover_consecutive_transfer: cover property (
     44      @(posedge vif.pclk) disable iff (!vif.preset_n)
     45      (vif.psel && vif.penable && vif.pready) ##1 (vif.psel && !vif.penable)
     46    );
     47  
     48  endmodule
