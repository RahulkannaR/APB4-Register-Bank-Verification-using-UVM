1  //-----------------------------------------------------------------------------
     2  // apb_coverage.sv
     3  // Functional coverage: register index, R/W type, PSLVERR, RO-vs-RW cross.
     4  // >>> NUM_REGS / READ_ONLY_MASK must match apb_reg_block.
     5  //-----------------------------------------------------------------------------
     6  `uvm_analysis_imp_decl(_cov)
     7  
     8  class apb_coverage extends uvm_component;
     9  
     10    `uvm_component_utils(apb_coverage)
     11  
     12    uvm_analysis_imp_cov #(apb_seq_item, apb_coverage) cov_imp;
     13  
     14    localparam int NUM_REGS      = 8;
     15    localparam int ADDR_OFFSET   = 4;
     16    localparam bit [NUM_REGS-1:0] READ_ONLY_MASK = 8'b0000_0001;
     17  
     18    apb_seq_item tr_cov;
     19    int          reg_idx;
     20    bit          is_ro;
     21  
     22    covergroup cg_apb;
     23      option.per_instance = 1;
     24  
     25      cp_reg_idx : coverpoint reg_idx {
     26        bins valid_idx[] = {[0:NUM_REGS-1]};
     27        bins out_of_range = {[NUM_REGS:$]};
     28      }
     29  
     30      cp_rw : coverpoint tr_cov.pwrite {
     31        bins write = {1};
     32        bins read  = {0};
     33      }
     34  
     35      cp_pslverr : coverpoint tr_cov.pslverr {
     36        bins ok    = {0};
     37        bins error = {1};
     38      }
     39  
     40      cp_is_ro : coverpoint is_ro {
     41        bins ro = {1};
     42        bins rw = {0};
     43      }
     44  
     45      // cross: every register hit for both read & write, split RO/RW
     46      x_idx_rw_ro : cross cp_reg_idx, cp_rw, cp_is_ro;
     47  
     48      // cross: confirm SLVERR seen on both mapped and unmapped paths
     49      x_idx_err : cross cp_reg_idx, cp_pslverr;
     50  
     51    endgroup
     52  
     53    function new(string name = "apb_coverage", uvm_component parent = null);
     54      super.new(name, parent);
     55      cg_apb = new();
     56    endfunction
     57  
     58    function void build_phase(uvm_phase phase);
     59      super.build_phase(phase);
     60      cov_imp = new("cov_imp", this);
     61    endfunction
     62  
     63    function void write_cov(apb_seq_item tr);
     64      tr_cov  = tr;
     65      reg_idx = tr.paddr / ADDR_OFFSET;
     66      is_ro   = (reg_idx < NUM_REGS) ? READ_ONLY_MASK[reg_idx] : 1'b0;
     67      cg_apb.sample();
     68    endfunction
     69  
     70  endclass
