     15  /// Address Decoder: Maps the input address combinatorially to an index.
     16  /// The address map `addr_map_i` is a packed array of rule_t structs.
     17  /// The ranges of any two rules may overlap. If so, the rule at the higher (more significant)
     18  /// position in `addr_map_i` prevails.
     19  ///
     20  /// There can be an arbitrary number of address rules. There can be multiple
     21  /// ranges defined for the same index. The start address has to be less than the end address.
     22  ///
     23  /// There is the possibility to add a default mapping:
     24  /// `en_default_idx_i`: Driving this port to `1'b1` maps all input addresses
     25  /// for which no rule in `addr_map_i` exists to the default index specified by
     26  /// `default_idx_i`. In this case, `dec_error_o` is always `1'b0`.
     27  ///
     28  /// The `Napot` parameter allows using naturally-aligned power of two (NAPOT) regions,
     29  /// using base addresses and masks instead of address ranges to specify rules.
     30  ///
     31  /// Assertions: The module checks every time there is a change in the address mapping
     32  /// if the resulting map is valid. It fatals if `start_addr` is higher than `end_addr` (non-NAPOT
     33  /// only) or if a mapping targets an index that is outside the number of allowed indices.
     34  /// It issues warnings if the address regions of any two mappings overlap (non-NAPOT only).
     35  
     36  `include "rtl/common_cells-master/src/cc_addr_decode_dync.sv"
     37  
     38  module cc_addr_decode #(
     39    /// Highest index which can happen in a rule.
     40    parameter int unsigned NoIndices = 32'd0,
     41    /// Total number of rules.
     42    parameter int unsigned NoRules   = 32'd1,
     43    /// Address type inside the rules and to decode.
     44    parameter type         addr_t    = logic,
     45    // Whether this is a NAPOT (base and mask) or regular range decoder
     46    parameter bit          Napot     = 0,
     47    /// The output index type `idx_t` can be specified either with the width `IdxWidth`
     48    /// or directly with the type `idx_t`. By default, it will use the maximum index
     49    /// `NoIndices` to calculate the required width.
     50    parameter int unsigned IdxWidth  = cc_pkg::idx_width(NoIndices),
     51    parameter type         idx_t     = logic [IdxWidth-1:0],
     52    /// Rule packed struct type.
     53    /// The address decoder expects three fields in `rule_t`:
     54    ///
     55    /// typedef struct packed {
     56    ///   idx_t  idx;
     57    ///   addr_t start_addr;
     58    ///   addr_t end_addr;
     59    /// } rule_t;
     60    ///
     61    ///  - `idx`:        The index to be returned for a matching rule. Usually an integer but can
     62    ///                  be any type of data.
     63    ///  - `start_addr`: start address of the range the rule describes, value is included in range
     64    ///  - `end_addr`:   end address of the range the rule describes, value is NOT included in range
     65    ///                  if `end_addr == '0` end of address space is assumed
     66    ///
     67    /// If `Napot` is 1, The field names remain the same, but the rule describes a naturally-aligned
     68    /// power of two (NAPOT) region instead of an address range: `start_addr` becomes the base address
     69    /// and `end_addr` the mask. See the wrapping module `cc_addr_decode_napot` for details.
     70    // verilog_lint: waive typedef-structs-unions
     71    parameter type         rule_t    = struct packed {
     72      idx_t  idx;
     73      addr_t start_addr;
     74      addr_t end_addr;
     75    }
     76  ) (
     77    /// Address to decode.
     78    input  addr_t               addr_i,
     79    /// Address map: rule with the highest array position wins on collision
     80    input  rule_t [NoRules-1:0] addr_map_i,
     81    /// Decoded index.
     82    output idx_t                idx_o,
     83    /// Decode is valid.
     84    output logic                dec_valid_o,
     85    /// Decode is not valid, no matching rule found.
     86    output logic                dec_error_o,
     87    /// Enable default port mapping.
     88    ///
     89    /// When not used, tie to `0`.
     90    input  logic                en_default_idx_i,
     91    /// Default port index.
     92    ///
     93    /// When `en_default_idx_i` is `1`, this will be the index when no rule matches.
     94    ///
     95    /// When not used, tie to `0`.
     96    input  idx_t                default_idx_i
     97  );
     98  
     99    // wraps the dynamic configuration version of the address decoder
     100    cc_addr_decode_dync #(
     101      .NoRules   ( NoRules   ),
     102      .addr_t    ( addr_t    ),
     103      .rule_t    ( rule_t    ),
     104      .Napot     ( Napot     ),
     105      .idx_t     ( idx_t     )
     106    ) i_addr_decode_dync (
     107      .addr_i,
     108      .addr_map_i,
     109      .idx_o,
     110      .dec_valid_o,
     111      .dec_error_o,
     112      .en_default_idx_i,
     113      .default_idx_i,
     114      .config_ongoing_i ( 1'b0 )
     115    );
     116  
     117  endmodule
